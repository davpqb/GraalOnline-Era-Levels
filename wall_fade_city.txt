// Wall fade system — "city" tile variant
// Inverted trigger logic from the white variant: these walls fade OUT
// when the player enters the flanking zones (used for exterior-facing walls).

dontblock();
drawoverplayer();

//#CLIENTSIDE
function onCreated(){
	this.imgwidth = getimgwidth(this.image) / 16;
	this.imgheight = getimgheight(this.image) / 16;
	this.mode = 1;
	onTimeOut();
}

function onTimeOut(){
	if(client.player_screenshotting){
		this.alpha = 1;
		return setTimer(0.01);
	}

	temp.inleft = player.x + 1 in |0, 23| && player.y + 1.5 in |49, 64|;
	temp.inright = player.x + 1 in |64-23, 64| && player.y + 1.5 in |49, 64|;

	if(inleft || inright) fadeOut();
	else fadeIn();

	setTimer(0.01);
}

function fadeOut(){
	if(this.alpha <= 0) return;
	this.alpha -= 0.1;
}

function fadeIn(){
	if(this.alpha >= 1) return;
	if(this.alpha < 1) this.alpha += 0.1;
}
